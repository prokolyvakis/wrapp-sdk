function consume(outcome: CreateOutcome): string {
  switch (outcome.kind) {
    case 'observed': return outcome.invoice.id;
    case 'pending': return outcome.invoiceId;
    case 'rejected': return outcome.referenceState;
    default: { const unreachable: never = outcome; return unreachable; }
  }
}
assert.equal(consume({ kind: 'rejected', errorCount: 1, rejectionSource: 'unknown', referenceState: 'unknown' }), 'unknown');
assert.equal(consume({ kind: 'pending', invoiceId: 'i', referenceState: 'unknown', identity: 'unavailable' }), 'i');

function status(outcome: InvoiceStatusOutcome): string {
  switch (outcome.kind) {
    case 'observed': return outcome.invoice.id + '/' + outcome.identity;
    case 'pending': return outcome.invoiceId + '/' + outcome.identity + '/' + (outcome.invoice?.my_data_uid ?? 'no-evidence');
    default: { const unreachable: never = outcome; return unreachable; }
  }
}
assert.equal(status({ kind: 'pending', invoiceId: 'i', referenceState: 'unknown', identity: 'unavailable' }), 'i/unavailable/no-evidence');

function route(event: VerifiedWebhook): string {
  switch (event.kind) {
    case 'invoice-observation': return event.invoice.id;
    case 'pdf': return event.eventTypeHint + ' ' + event.downloadUrl;
    case 'pos-payment-error': return event.providerMessage;
    default: { const unreachable: never = event; return unreachable; }
  }
}

  signature: createHmac('sha256', 'synthetic-consumer-key').update(webhookBody).digest('hex'),
  keys: ['synthetic-consumer-key'],
  eventType: 'thermal-print-pdf',
});
assert.equal(route(webhook), 'thermal-print-pdf https://example.invalid/pdf');
assert.equal(webhook.eventTypeAuthenticated, false);

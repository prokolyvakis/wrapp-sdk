function describeEvidence(node: ProviderJson | undefined): string {
  if (node === undefined) return 'absent';
  switch (node.kind) {
    case 'null': return 'null';
    case 'boolean': return 'boolean:' + String(node.value);
    case 'string': return 'string:' + node.value;
    case 'number': return 'number:' + node.text;
    case 'array': return 'array(' + node.items.map(describeEvidence).join(',') + ')';
    case 'object': return 'object(' + node.entries.map((entry) => entry.key + '=' + describeEvidence(entry.value)).join(',') + ')';
  }
}
function consume(outcome: CreateOutcome): string {
  switch (outcome.kind) {
    case 'observed': return outcome.invoice.id;
    case 'pending': return outcome.invoiceId;
    case 'rejected': return outcome.referenceState;
    default: { const unreachable: never = outcome; return unreachable; }
  }
}

function status(outcome: InvoiceStatusOutcome): string {
  switch (outcome.kind) {
    case 'observed': return outcome.invoice.id + '/' + outcome.identity;
    case 'pending': return outcome.invoiceId + '/' + outcome.identity + '/' + (outcome.invoice?.my_data_uid ?? 'no-evidence');
    default: { const unreachable: never = outcome; return unreachable; }
  }
}
